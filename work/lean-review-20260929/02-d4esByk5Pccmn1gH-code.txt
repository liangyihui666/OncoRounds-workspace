function main({ raw_json }) {
  const input = String(raw_json ?? "").trim();

  if (!input) {
    throw new Error("病例结构化提取结果为空");
  }

  /**
   * 删除模型可能输出的 Markdown 代码围栏。
   */
  function removeCodeFence(value) {
    return String(value ?? "")
      .replace(/^\s*```(?:json)?\s*/i, "")
      .replace(/\s*```\s*$/i, "")
      .trim();
  }

  /**
   * 截取第一个 JSON 根对象。
   */
  function extractJsonObject(value) {
    const source = removeCodeFence(value);
    const start = source.indexOf("{");
    const end = source.lastIndexOf("}");

    if (start < 0 || end <= start) {
      throw new Error("未找到可解析的 JSON 根对象");
    }

    return source.slice(start, end + 1);
  }

  /**
   * 修复 JSON 字符串内部未转义的半角双引号，
   * 同时处理字符串内部的真实换行和制表符。
   */
  function repairUnescapedQuotes(value) {
    let result = "";
    let inString = false;
    let escaped = false;

    for (let index = 0; index < value.length; index++) {
      const char = value[index];

      if (!inString) {
        result += char;

        if (char === '"') {
          inString = true;
          escaped = false;
        }

        continue;
      }

      if (escaped) {
        result += char;
        escaped = false;
        continue;
      }

      if (char === "\\") {
        result += char;
        escaped = true;
        continue;
      }

      if (char === "\r") {
        if (value[index + 1] === "\n") {
          index += 1;
        }

        result += "\\n";
        continue;
      }

      if (char === "\n") {
        result += "\\n";
        continue;
      }

      if (char === "\t") {
        result += "\\t";
        continue;
      }

      if (char !== '"') {
        result += char;
        continue;
      }

      // 判断当前双引号是字符串结束符，还是正文内部未转义引号。
      let nextIndex = index + 1;

      while (
        nextIndex < value.length &&
        /\s/.test(value[nextIndex])
      ) {
        nextIndex += 1;
      }

      const nextChar =
        nextIndex < value.length ? value[nextIndex] : "";

      const isClosingQuote =
        nextChar === ":" ||
        nextChar === "," ||
        nextChar === "}" ||
        nextChar === "]" ||
        nextChar === "";

      if (isClosingQuote) {
        result += '"';
        inString = false;
      } else {
        result += '\\"';
      }
    }

    if (inString) {
      throw new Error("JSON 字符串引号未闭合，无法安全修复");
    }

    return result;
  }

  /**
   * 删除对象或数组末尾多余的逗号。
   */
  function removeTrailingCommas(value) {
    let result = "";
    let inString = false;
    let escaped = false;

    for (let index = 0; index < value.length; index++) {
      const char = value[index];

      if (inString) {
        result += char;

        if (escaped) {
          escaped = false;
        } else if (char === "\\") {
          escaped = true;
        } else if (char === '"') {
          inString = false;
        }

        continue;
      }

      if (char === '"') {
        inString = true;
        result += char;
        continue;
      }

      if (char === ",") {
        let nextIndex = index + 1;

        while (
          nextIndex < value.length &&
          /\s/.test(value[nextIndex])
        ) {
          nextIndex += 1;
        }

        const nextChar =
          nextIndex < value.length ? value[nextIndex] : "";

        if (nextChar === "}" || nextChar === "]") {
          continue;
        }
      }

      result += char;
    }

    return result;
  }

  /**
   * 解析原始 JSON；失败时依次进行受控修复。
   */
  const source = extractJsonObject(input);
  let parsed;

  try {
    parsed = JSON.parse(source);
  } catch (firstError) {
    const quoteRepaired = repairUnescapedQuotes(source);
    const fullyRepaired = removeTrailingCommas(quoteRepaired);

    try {
      parsed = JSON.parse(fullyRepaired);
    } catch (secondError) {
      throw new Error(
        `病例结构化 JSON 修复失败：${secondError.message}`
      );
    }
  }

  if (
    !parsed ||
    typeof parsed !== "object" ||
    Array.isArray(parsed)
  ) {
    throw new Error("病例结构化结果必须是单个 JSON 对象");
  }

  // 格式修复不改写日期、病史、不确定性或已解析字段。
  return {
    repaired_json: JSON.stringify(parsed),
    repaired_data: parsed
  };
}
