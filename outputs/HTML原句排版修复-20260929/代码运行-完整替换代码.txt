// Conservative display normalization with offsets back to the unmodified source.
// Cell/row boundaries are not removed. English word boundaries and medical signs remain.
function sourceLayout(value) {
  const s=String(value||''),tokens=[];
  const emit=(text,start,end)=>{for(const c of text)tokens.push({c,start,end});};
  for(let i=0;i<s.length;){
    const tail=s.slice(i);let m;
    if((m=tail.match(/^<br\s*\/?\s*>/i))){emit(' ',i,i+m[0].length);i+=m[0].length;continue;}
    if((m=tail.match(/^<\/?(?:b|strong|em|i|span)(?:\s[^>]*)?>/i))){i+=m[0].length;continue;}
    if((m=tail.match(/^&(?:nbsp|amp|lt|gt|quot|apos);/))){const map={'&nbsp;':' ','&amp;':'&','&lt;':'<','&gt;':'>','&quot;':'"','&apos;':"'"};emit(map[m[0]],i,i+m[0].length);i+=m[0].length;continue;}
    if((m=tail.match(/^\\+(\$|%|~|sim\b)/))){emit(m[1]==='$'?'':m[1]==='sim'?'~':m[1],i,i+m[0].length);i+=m[0].length;continue;}
    emit(s[i],i,i+1);i++;
  }
  const cjk=c=>/[\u3400-\u9fff]/.test(c||''),flat=[];
  for(let i=0;i<tokens.length;i++){
    if(/\s/.test(tokens[i].c)){
      let j=i+1;while(j<tokens.length&&/\s/.test(tokens[j].c))j++;
      if(!(cjk(flat[flat.length-1]?.c)&&cjk(tokens[j]?.c))&&flat.length)flat.push({c:' ',start:tokens[i].start,end:tokens[j-1].end});
      i=j-1;
    }else flat.push(tokens[i]);
  }
  if(flat[flat.length-1]?.c===' ')flat.pop();
  return {text:flat.map(t=>t.c).join(''),tokens:flat};
}

function sourceSection(cleaned,label){
  const s=String(cleaned||''),start=s.indexOf('【'+label+'】');if(start<0)return '';
  const tail=s.slice(start+label.length+2),end=tail.search(/【(?:来源类型|清洗状态|患者病例正文|文章导语与二次资料|专家观点|图片索引|冲突与不确定项|敏感信息提示|已移除内容类别|来源说明)】/);
  return (end<0?tail:tail.slice(0,end)).trim();
}
// Only for locating patient passages, never for rewriting their clinical content.
// Retain offsets into the source and restore the source passage after matching.
function caseSourceLayout(value){
  const base=sourceLayout(value),tokens=[];
  for(let i=0;i<base.tokens.length;i++){
    const t=base.tokens[i];
    if(t.c===' '&&base.tokens[i+1]?.c==='。')continue;
    tokens.push({...t,c:/[“”]/.test(t.c)?'"':t.c});
  }
  return {text:tokens.map(t=>t.c).join(''),tokens};
}
function checkCaseSource(cleaned,documentText){
  const state=sourceSection(cleaned,'清洗状态').replace(/[。；;\s]/g,'');
  if(state!=='clean')throw new Error('CASE_SOURCE_REVIEW_REQUIRED：清洗状态为'+(state||'缺失')+'，应先核对原文，不进入出题。');
  const body=sourceSection(cleaned,'患者病例正文');
  if(body.length<20||/^(?:无|未提供|未找到)/.test(body))throw new Error('CASE_SOURCE_EMPTY：未提取到患者正文。');
  const source=String(documentText||''),original=caseSourceLayout(source);
  if(!original.text)throw new Error('CASE_SOURCE_EMPTY：缺少文档解析原文。');
  const lines=body.split(/\r?\n/).map(s=>s.trim()).filter(Boolean),matches=[],unmatched=[];
  for(let i=0;i<lines.length;i++){
    const needle=caseSourceLayout(lines[i]).text,at=needle?original.text.indexOf(needle):-1;
    if(at<0){unmatched.push(i+1);continue;}
    const start=original.tokens[at].start,end=original.tokens[at+needle.length-1].end;
    const text=source.slice(start,end);
    matches.push({line:i+1,start,end,text,match_mode:text===lines[i]?'exact':'display_layout_only'});
  }
  if(unmatched.length)throw new Error('CASE_SOURCE_REWRITE：清洗正文有'+unmatched.length+'段未能回溯文档原句（段号'+unmatched.join(',')+'）。已兼容中文双引号和句号前空格；请检查文字、数值或段落拼接差异。');
  return {body:matches.map(m=>m.text).join('\n\n'),report:{
    scope:'literal_source_trace_not_medical_validation',status:'PASS',normalization_policy:'case-display-layout-1.2',
    patient_paragraph_count:lines.length,matched_paragraph_count:matches.length,
    display_normalized_paragraph_count:matches.filter(m=>m.match_mode==='display_layout_only').length,
    source_restored:true,paragraph_matches:matches.map(({text,...m})=>m),
    semantic_support:'not_independently_verified'
  }};
}

function classifyTumor(value) {
  const second = typeof value === 'string' ? value.trim() : '';
  if (!second || !Object.prototype.hasOwnProperty.call(TUMOR_DISPLAY_MAP, second))
    return { cancer_type:'', second_cancer_type:'', taxonomy_warnings:[second ? '二级癌种不在CSV字典：'+second : '未确定原发肿瘤对应的二级癌种'] };
  const first = TUMOR_DISPLAY_MAP[second];
  return { cancer_type:first, second_cancer_type:second, taxonomy_warnings:first ? [] : ['二级癌种已确定，但一级归属尚未确认：'+second] };
}
function buildOriginalCase({ case_json }) {
  const raw = String(case_json ?? "").trim();
  const start = raw.indexOf("{");
  const end = raw.lastIndexOf("}");

  if (start < 0 || end <= start) {
    throw new Error("病例结构化提取结果不是可解析的 JSON 对象");
  }

  const data = JSON.parse(raw.slice(start, end + 1));
  const requiredFields = [
    "case_title",
    "disease"
  ];
  const missingFields = requiredFields.filter(
    (field) => !String(data[field] ?? "").trim()
  );

  if (missingFields.length > 0) {
    throw new Error(`缺少生产工作流必需字段: ${missingFields.join(", ")}`);
  }

  function isSafeRemoteImageUrl(value) {
    if (!value) return false;

    try {
      const url = new URL(value);
      const host = url.hostname.toLowerCase();
      const privateIpv4 = /^(10\.|127\.|169\.254\.|192\.168\.|172\.(1[6-9]|2\d|3[01])\.)/;
      const sensitiveQueryName = /(^|_)(token|access_token|api_key|apikey|auth|authorization|credential|signature|sig)($|_)/i;

      if (!/^https?:$/.test(url.protocol)) return false;
      if (url.username || url.password) return false;
      if (host === "localhost" || host.endsWith(".local") || privateIpv4.test(host)) return false;
      if ([...url.searchParams.keys()].some((key) => sensitiveQueryName.test(key))) return false;

      return true;
    } catch {
      return false;
    }
  }

  function parseCaseImageIndex(value) {
    const rawIndex = String(value ?? "").trim();
    if (!rawIndex) {
      return { case_image_index: "", case_intro_images: [] };
    }

    const normalizedLines = [];
    const remoteUrls = [];

    for (const rawLine of rawIndex.split(/\r?\n/)) {
      const line = rawLine.trim();
      if (!line) continue;

      const parts = line.split("|").map((part) => part.trim());
      const imageId = parts[0] || "";
      const sourceUrl = parts[1] || "";
      const timepoint = parts[2] || "";
      const caption = parts.slice(3).join(" | ").trim();

      if (!/^IMG\d+$/.test(imageId)) continue;

      const safeUrl = isSafeRemoteImageUrl(sourceUrl) ? sourceUrl : "";
      normalizedLines.push(`${imageId} | ${safeUrl} | ${timepoint} | ${caption}`);

      if (safeUrl && !remoteUrls.includes(safeUrl)) {
        remoteUrls.push(safeUrl);
      }
    }

    return {
      case_image_index: normalizedLines.join("\n"),
      case_intro_images: remoteUrls
    };
  }

  const imageData = parseCaseImageIndex(data.case_image_index);

  function stableCaseId(title) {
    const s = String(title || "").trim();
    if (!s) return "";
    let h = 5381;
    for (let i = 0; i < s.length; i++) {
      h = ((h << 5) + h + s.charCodeAt(i)) >>> 0;
    }
    return "CASE-" + h.toString(16).toUpperCase();
  }

  const classification = classifyTumor(data.second_cancer_type);
  const production_variables = {
    ...classification,
    case_id: data.case_id || stableCaseId(data.case_title),
    case_title: data.case_title || "",
    disease: data.disease || "",
    case_context: data.case_context || "",
    learning_points: data.learning_points_candidate || "",
    question_count: 5,
    difficulty: data.difficulty || "进阶",
    case_mode: data.case_mode || "全程管理",
    case_timeline: data.case_timeline || "",
    biomarkers: data.biomarkers || "",
    decision_points: data.decision_points_candidate || "",
    adverse_events: data.adverse_events || "",
    evidence_notes: [data.evidence_notes || "", ...classification.taxonomy_warnings].filter(Boolean).join("\n"),
    case_image_index: imageData.case_image_index,
    case_intro_images: imageData.case_intro_images
  };

  return {production_variables,...production_variables};
}

function main(args) {
 const trace=checkCaseSource(args.cleaned_text,args.document_text);
 const result=buildOriginalCase(args);
 const source=result.production_variables;
 if(!source||typeof source!=='object') throw new Error('病例字段整理未返回有效对象。');
 const extraction_check=trace.report;
 source.case_source_text=trace.body;
 if(args.source_id){source.case_id=args.source_id;result.case_id=source.case_id;}
 source.evidence_notes=[source.evidence_notes||'','【事实来源】case_source_text保留患者原文。结构化字段仅作索引；疑似、时间点、分期冲突须回原文核对。'].filter(Boolean).join('\n');
 result.evidence_notes=source.evidence_notes;
 const fields=['case_id','case_title','disease','case_context','learning_points','case_mode','case_timeline','biomarkers','decision_points','adverse_events','evidence_notes','case_intro_images','cancer_type','second_cancer_type','case_source_text'];
 const payload={};
 for(const key of fields){
   if(key==='case_intro_images'){payload[key]=[];continue;}
   if(source[key]!=null&&typeof source[key]!=='string')throw new Error('病例字段类型错误：'+key);
   payload[key]=source[key]??'';
 }
 return {...result,extraction_check,case_payload_json:JSON.stringify(payload)};
}
